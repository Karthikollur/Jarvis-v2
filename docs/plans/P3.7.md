`run_named_script` inside `mutating` (replaces the body; result shape unchanged):

```python
name = args["name"]

job = _kernel32.CreateJobObjectW(None, None)
if not job:
    raise ctypes.WinError(ctypes.get_last_error())
try:
    with tempfile.TemporaryFile() as out:  # a file, not a pipe: nothing waits for EOF
        process = subprocess.Popen(
            script.command, shell=False, cwd=script.cwd,
            stdin=subprocess.DEVNULL, stdout=out, stderr=subprocess.STDOUT,
            creationflags=subprocess.CREATE_NO_WINDOW | subprocess.CREATE_NEW_PROCESS_GROUP,
        )
        # ponytail: a grandchild started before this line escapes the job (microseconds).
        if not _kernel32.AssignProcessToJobObject(job, int(process._handle)):
            process.kill()
            process.wait(timeout=5)
            raise ctypes.WinError(ctypes.get_last_error())
        try:
            code = process.wait(timeout=script.timeout)
        except subprocess.TimeoutExpired:
            code = None
        _kernel32.TerminateJobObject(job, 1)  # the whole tree, also after a normal exit (D4)
        process.wait(timeout=5)
        out.seek(0)
        output = _tail(out.read())
finally:
    _kernel32.CloseHandle(job)
return {"ran": name, "timed_out": code is None, "exit_code": code, "output": output}
```

## Section F (optional) — undo retries a held rename (commit `P3.7: undo retries a held rename`)

```python
def _rename(source: Path, target: Path, tries: int = 3, pause: float = 0.2) -> None:
    for attempt in range(tries):
        try:
            os.rename(source, target)
            return
        except PermissionError as error:  # WinError 5/32: a handle held inside (indexer, antivirus, editor)
            if attempt + 1 < tries:
                time.sleep(pause)
                continue
            raise UndoRefused(f"move failed: {type(error).__name__}") from error
        except OSError as error:
            if getattr(error, "winerror", None) == 17 or error.errno == errno.EXDEV:
                raise UndoRefused("cannot move across drives") from error
            raise UndoRefused(f"move failed: {type(error).__name__}") from error
```
